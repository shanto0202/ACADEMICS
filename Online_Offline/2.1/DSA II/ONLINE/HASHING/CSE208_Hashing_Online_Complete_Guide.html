Offline Foundation

1. Assignment-এর exact core

তোমার offline hash table-এ মূলত তিনটি collision-resolution method থাকতে হবে:

Method Storage Probe/position Deletion issue
Chaining প্রতি bucket-এ linked list hash(key) % N list থেকে erase
Double hashing এক slot-এ এক item (h(key)+i*aux(key))%N tombstone দরকার
Custom probing এক slot-এ এক item (h(key)+C1*i*aux(key)+C2*i*i)%N tombstone দরকার

Initial table size 13, load factor 0.5-এর বেশি হলে nextPrime(2*N)-এ expansion; 0.25-এর নিচে হলে previousPrime(N/2)-এ compaction, কিন্তু initial size-এর নিচে নয়। Resize করার সময় পুরোনো index copy করা যাবে না—প্রতিটি live key নতুন capacity দিয়ে আবার hash করতে হবে।

2. Safe hash helpers

using ull = unsigned long long;

ull hash1(const string &s) {              // polynomial rolling
    ull h = 0;
    for (unsigned char c : s) h = h * 31 + c;
    return h;
}

ull hash2(const string &s) {              // FNV-1a 64 bit
    ull h = 14695981039346656037ULL;
    for (unsigned char c : s) {
        h ^= c;
        h *= 1099511628211ULL;
    }
    return h;
}

int auxiliaryHash(const string &s, int m) {
    ull h = 0;
    for (unsigned char c : s) h = h * 41 + c;
    return 1 + h % (m - 1);
}

1 + ... % (m-1) step-কে 1..m-1 রাখে; step কখনো 0 হয় না। m prime হলে প্রতিটি non-zero step-এর সাথে gcd(step,m)=1, তাই double hashing সর্বোচ্চ mটি distinct slot visit করতে পারে।

3. Collision, hit, duplicate—এক জিনিস নয়

  • Duplicate: exact key আগেই আছে; সাধারণত discard/update হবে, নতুন collision count নয়।
  • Chaining collision: নতুন unique key-এর home bucket আগে থেকেই non-empty হলে 1 collision (teacher যদি node-comparison count চান, সেই convention আলাদা করে follow করবে)।
  • Open addressing collision: insertion-এর সময় প্রতিটি occupied non-matching slot encounter সাধারণত 1 collision।
  • Hit/probe: search-এ যতগুলো table slot বা list node access হয়েছে।
  • Tombstone: deleted slot; search এখানে থামবে না, insertion এটিকে reuse করতে পারবে।

4. Resizing invariants

void rehash(int newSize) {
    auto old = table;
    table.assign(newSize, Slot{});
    itemCount = 0;
    for (auto &slot : old)
        if (slot.state == OCCUPIED)
            insertWithoutResize(slot.key, slot.value);
}

Rehash-এর সময় public insert() call করলে collision/report/counter ভুলভাবে বাড়তে পারে এবং recursive resize হতে পারে। তাই আলাদা insertWithoutResize() বা insertRaw() রাখবে।

Exam 1 — Priority-weighted sorted chaining (Section B, 30 min)

Question summary

  • Fixed N = 13, resize নয়।
  • Hash1(k) = Σ(i × ASCII(k[i])) mod 13, position 1-indexed।
  • weight = ((ASCII(first)+ASCII(last)) % 100) / 100.0।
  • Chain descending weight; equal weight হলে later insertion পরে থাকবে।
  • Duplicate discard।

কী কী change হবে

  1. hash1() replace। 2. Node-এ double weight যোগ। 3. chaining insert-এর push_back বদলে ordered position-এ insert। 4. print-এ weight দেখানো।

Complete solution

#include <bits/stdc++.h>
using namespace std;

struct Node {
    string key;
    int value;
    double weight;
};

class HashTable {
    static const int N = 13;
    vector<list<Node>> table;
    int collisions = 0;

    int hash1(const string &s) {
        long long h = 0;
        for (int i = 0; i < (int)s.size(); i++)
            h += 1LL * (i + 1) * (unsigned char)s[i];
        return h % N;
    }

    double getWeight(const string &s) {
        return (((unsigned char)s.front() +
                 (unsigned char)s.back()) % 100) / 100.0;
    }

public:
    HashTable() : table(N) {}

    bool insert(const string &key, int value) {
        int idx = hash1(key);
        auto &chain = table[idx];

        for (const Node &x : chain)
            if (x.key == key) return false;       // duplicate discard

        if (!chain.empty()) collisions++;
        Node cur{key, value, getWeight(key)};

        auto it = chain.begin();
        while (it != chain.end() && it->weight >= cur.weight)
            ++it;                                 // equal হলে old-এর পরে
        chain.insert(it, cur);
        return true;
    }

    void print() {
        cout << fixed << setprecision(2);
        for (int i = 0; i < N; i++) {
            cout << "Bucket " << i << ": ";
            if (table[i].empty()) cout << "(empty)";
            bool first = true;
            for (const Node &x : table[i]) {
                if (!first) cout << " -> ";
                cout << x.key << " (" << x.weight << ")";
                first = false;
            }
            cout << '\n';
        }
    }
};

int main() {
    vector<string> words = {"cat","dog","bat","rat","sun","fun",
                            "run","top","pot","opt","art","tar"};
    HashTable ht;
    for (int i = 0; i < (int)words.size(); i++)
        ht.insert(words[i], i + 1);
    ht.print();
}

Why the loop works: যতক্ষণ current node-এর weight নতুন node-এর চেয়ে বড় বা equal, iterator সামনে যায়। প্রথম smaller weight-এর আগে new node ঢুকে যায়। তাই descending order বজায় থাকে; >= থাকার কারণে equal weight-এর নতুন item পরে যায়। Complexity: average insertion O(chain length)।

Exam 2 — Print complete probe sequence (B1+B2, 30 min)

Question summary

Open addressing table-এ একটি printProbeSequence(key) method যোগ করতে হবে। Current table state-এ search যে slotগুলো visit করত, key পাওয়া বা first never-used EMPTY slot পাওয়া পর্যন্ত সেগুলো print করবে। Chaining-এ এটি applicable নয়।

Complete reusable solution

#include <bits/stdc++.h>
using namespace std;
using ull = unsigned long long;

enum State { EMPTY, OCCUPIED, DELETED };
enum Method { DOUBLE_HASHING, CUSTOM_PROBING };

struct Slot {
    string key;
    int value = 0;
    State state = EMPTY;
};

class HashTable {
    vector<Slot> table;
    Method method;
    const int C1 = 3, C2 = 5;

    ull h1(const string &s) const {
        ull h = 0;
        for (unsigned char c : s) h = h * 31 + c;
        return h;
    }
    ull aux(const string &s) const {
        ull h = 0;
        for (unsigned char c : s) h = h * 41 + c;
        return 1 + h % (table.size() - 1);
    }
    int probe(const string &key, int i) const {
        ull m = table.size(), a = h1(key) % m, step = aux(key);
        if (method == DOUBLE_HASHING)
            return (a + 1ULL * i * step) % m;
        return (a + 1ULL * C1 * i * step + 1ULL * C2 * i * i) % m;
    }

public:
    HashTable(int n, Method md) : table(n), method(md) {}

    bool insert(const string &key, int value) {
        int firstDeleted = -1;
        for (int i = 0; i < (int)table.size(); i++) {
            int idx = probe(key, i);
            if (table[idx].state == OCCUPIED && table[idx].key == key) {
                table[idx].value = value;
                return true;
            }
            if (table[idx].state == DELETED && firstDeleted == -1)
                firstDeleted = idx;
            if (table[idx].state == EMPTY) {
                if (firstDeleted != -1) idx = firstDeleted;
                table[idx] = {key, value, OCCUPIED};
                return true;
            }
        }
        if (firstDeleted != -1) {
            table[firstDeleted] = {key, value, OCCUPIED};
            return true;
        }
        return false;
    }

    void printProbeSequence(const string &key) const {
        for (int i = 0; i < (int)table.size(); i++) {
            int idx = probe(key, i);
            if (i) cout << " -> ";
            cout << idx;
            if (table[idx].state == EMPTY ||
               (table[idx].state == OCCUPIED && table[idx].key == key)) break;
        }
        cout << '\n';
    }
};

int main() {
    int m = 20011;                         // 10000 keys-এর জন্য prime capacity
    HashTable ht(m, DOUBLE_HASHING);
    // এখানে offline-এর generated 10000 words insert করবে
    int n; cin >> n;
    while (n--) {
        string key; cin >> key;
        ht.printProbeSequence(key);
    }
}

Common mistake: fixed পাঁচটি index print করা নয়। Current state অনুযায়ী first EMPTY/matching key পর্যন্ত যাবে; DELETED পেলে continue করবে।

Exam 3 — Set intersection, union, difference (C1+C2, 30 min)

Idea

নিজের chaining hash set-এ A ও B insert করি। তারপর unique values scan করে contains() দিয়ে তিনটি result বানাই। Expected time O(n+m); order matter করে না।

Complete solution (own hash table, no unordered_set shortcut)

#include <bits/stdc++.h>
using namespace std;

class IntHashSet {
    vector<list<int>> t;
    int h(int x) const {
        long long y = x;
        y ^= (y >> 16); y *= 0x45d9f3b; y ^= (y >> 16);
        return (y % (long long)t.size() + t.size()) % t.size();
    }
public:
    IntHashSet(int n = 1009) : t(n) {}
    bool contains(int x) const {
        int id = h(x);
        for (int y : t[id]) if (y == x) return true;
        return false;
    }
    bool insert(int x) {
        if (contains(x)) return false;
        t[h(x)].push_back(x);
        return true;
    }
};

void printLine(const string &name, const vector<int> &v) {
    cout << name << ":";
    for (int x : v) cout << ' ' << x;
    cout << '\n';
}

int main() {
    ios::sync_with_stdio(false); cin.tie(nullptr);
    int n, m; cin >> n;
    vector<int> A(n); for (int &x : A) cin >> x;
    cin >> m;
    vector<int> B(m); for (int &x : B) cin >> x;

    IntHashSet a(2003), b(2003), seen(2003);
    for (int x : A) a.insert(x);
    for (int x : B) b.insert(x);

    vector<int> inter, uni, diff;
    for (int x : A) {
        if (seen.insert(x)) {
            uni.push_back(x);
            if (b.contains(x)) inter.push_back(x);
            else diff.push_back(x);
        }
    }
    for (int x : B) if (seen.insert(x)) uni.push_back(x);

    printLine("Intersection", inter);
    printLine("Union", uni);
    printLine("Diff (A-B)", diff);
}

Exam 4 — Generic two-layer hash table (B2, 35 min)

Structure

Outer table key1 দিয়ে group খুঁজে পায়; প্রতিটি outer group-এর নিজের inner table আছে, যেখানে key2 → value রাখা হয়। Chaining নিলে collision handling সহজ ও safe।

Complete solution

#include <bits/stdc++.h>
using namespace std;

class InnerTable {
    vector<list<pair<string,int>>> t;
    size_t h(const string &s) const { return hash<string>{}(s) % t.size(); }
public:
    InnerTable(int n = 101) : t(n) {}
    void insert(const string &k, int v) {
        auto &b = t[h(k)];
        for (auto &p : b) if (p.first == k) { p.second = v; return; }
        b.push_back({k, v});
    }
    bool find(const string &k, int &v) const {
        for (auto &p : t[h(k)]) if (p.first == k) { v = p.second; return true; }
        return false;
    }
    vector<pair<string,int>> all() const {
        vector<pair<string,int>> ans;
        for (auto &b : t) for (auto &p : b) ans.push_back(p);
        return ans;
    }
};

struct Group {
    string key1;
    InnerTable inner;
    Group(string k, int n) : key1(move(k)), inner(n) {}
};

class TwoLayerHash {
    int N;
    vector<list<Group>> outer;
    size_t h(const string &s) const { return hash<string>{}(s) % N; }

    Group* getGroup(const string &k) {
        for (auto &g : outer[h(k)]) if (g.key1 == k) return &g;
        return nullptr;
    }
public:
    TwoLayerHash(int n = 101) : N(n), outer(n) {}
    void insert(const string &k1, const string &k2, int value) {
        Group *g = getGroup(k1);
        if (!g) {
            outer[h(k1)].emplace_back(k1, N);
            g = &outer[h(k1)].back();
        }
        g->inner.insert(k2, value);
    }
    void find(const string &k1, const string &k2) {
        Group *g = getGroup(k1); int value;
        if (g && g->inner.find(k2, value)) cout << value << '\n';
        else cout << "null\n";
    }
    void find(const string &k1) {
        Group *g = getGroup(k1);
        if (!g) { cout << "null\n"; return; }
        auto v = g->inner.all();
        for (auto &[k, val] : v) cout << '(' << k << ", " << val << ") ";
        cout << '\n';
    }
};

int main() {
    TwoLayerHash table(101);
    table.insert("bd", "dhaka", 500);
    table.insert("bd", "ctg", 300);
    table.insert("uk", "london", 800);
    table.find("bd", "dhaka");
    table.find("bd", "syl");
    table.find("bd");
    table.find("china");
}

Exam 5 — Two-layer permission database, mixed probing (Shift 2)

Required design

  • Outer: Hash1(group_id) + linear probing।
  • Inner: Hash2(user_id) + quadratic probing h+i²।
  • SEARCH group user, SEARCH group, DELETE group user।
  • Inner deletion-এ tombstone বাধ্যতামূলক।

Complete solution

#include <bits/stdc++.h>
using namespace std;

enum State { EMPTY, OCCUPIED, DELETED };

struct UserSlot {
    string user, permission;
    State state = EMPTY;
};

class InnerTable {
    vector<UserSlot> t;
    size_t h2(const string &s) const {
        unsigned long long h = 14695981039346656037ULL;
        for (unsigned char c : s) { h ^= c; h *= 1099511628211ULL; }
        return h % t.size();
    }
public:
    InnerTable(int n = 1) : t(n) {}
    bool insert(const string &u, const string &p) {
        int firstDeleted = -1, n = t.size();
        for (int i = 0; i < n; i++) {
            int id = (h2(u) + 1LL * i * i) % n;
            if (t[id].state == OCCUPIED && t[id].user == u) {
                t[id].permission = p; return true;
            }
            if (t[id].state == DELETED && firstDeleted == -1) firstDeleted = id;
            if (t[id].state == EMPTY) {
                if (firstDeleted != -1) id = firstDeleted;
                t[id] = {u, p, OCCUPIED}; return true;
            }
        }
        if (firstDeleted != -1) { t[firstDeleted] = {u,p,OCCUPIED}; return true; }
        return false;
    }
    bool search(const string &u, string &p) const {
        int n = t.size();
        for (int i = 0; i < n; i++) {
            int id = (h2(u) + 1LL * i * i) % n;
            if (t[id].state == EMPTY) return false;
            if (t[id].state == OCCUPIED && t[id].user == u) {
                p = t[id].permission; return true;
            }
        }
        return false;
    }
    bool erase(const string &u, string &p) {
        int n = t.size();
        for (int i = 0; i < n; i++) {
            int id = (h2(u) + 1LL * i * i) % n;
            if (t[id].state == EMPTY) return false;

Exam 6 — Double hashing rehash by load factor (A1/B1, 30 min)

Rules

  • Input initial N and maximum load L।
  • Insert-এর পরে load > L হলে approximately double prime size।
  • Delete-এর পরে load < 0.3L হলে approximately half prime size; initial N-এর নিচে নয়।

Complete solution core

#include <bits/stdc++.h>
using namespace std;
using ull = unsigned long long;

bool isPrime(int n) {
    if (n < 2) return false;
    for (int i = 2; 1LL*i*i <= n; i++) if (n%i == 0) return false;
    return true;
}
int nextPrime(int n) { while (!isPrime(n)) ++n; return n; }
int prevPrime(int n) { while (n >= 2 && !isPrime(n)) --n; return n; }

enum State { EMPTY, OCCUPIED, DELETED };
struct Slot { string key; int value=0; State state=EMPTY; };

class DoubleHashTable {
    vector<Slot> t;
    int items = 0, initialSize;
    double L;

    ull H(const string &s) const {
        ull h=0; for (unsigned char c:s) h=h*31+c; return h;
    }
    int step(const string &s) const {
        ull h=0; for (unsigned char c:s) h=h*41+c;
        return 1 + h%(t.size()-1);
    }
    int pos(const string &s,int i) const {
        return (H(s)%t.size() + 1ULL*i*step(s))%t.size();
    }
    bool insertRaw(const string &key,int value) {
        int firstDeleted=-1;
        for (int i=0;i<(int)t.size();i++) {
            int id=pos(key,i);
            if (t[id].state==OCCUPIED && t[id].key==key) {
                t[id].value=value; return false;
            }
            if (t[id].state==DELETED && firstDeleted==-1) firstDeleted=id;
            if (t[id].state==EMPTY) {
                if (firstDeleted!=-1) id=firstDeleted;
                t[id]={key,value,OCCUPIED}; items++; return true;
            }
        }
        if (firstDeleted!=-1) {
            t[firstDeleted]={key,value,OCCUPIED}; items++; return true;
        }
        return false;
    }
    void rehash(int newSize) {
        vector<Slot> old=move(t);
        t.assign(newSize,{}); items=0;
        for (auto &x:old) if (x.state==OCCUPIED) insertRaw(x.key,x.value);
        cout << "Rehashed: " << old.size() << " -> " << t.size() << '\n';
    }
public:
    DoubleHashTable(int N,double maxLoad)
        : initialSize(nextPrime(N)), L(maxLoad) { t.resize(initialSize); }

    void insert(const string &key,int value) {
        bool added=insertRaw(key,value);
        if (!added) return;
        if ((double)items/t.size()>L) rehash(nextPrime(2*t.size()));
    }
    bool erase(const string &key) {
        for (int i=0;i<(int)t.size();i++) {
            int id=pos(key,i);
            if (t[id].state==EMPTY) return false;
            if (t[id].state==OCCUPIED && t[id].key==key) {
                t[id].state=DELETED; items--;
                if ((double)items/t.size()<0.3*L && (int)t.size()>initialSize) {
                    int ns=max(initialSize,prevPrime(t.size()/2));
                    if (ns<t.size()) rehash(ns);
                }
                return true;
            }
        }
        return false;
    }
    bool search(const string &key,int &value) const {
        for (int i=0;i<(int)t.size();i++) {
            int id=pos(key,i);
            if (t[id].state==EMPTY) return false;
            if (t[id].state==OCCUPIED && t[id].key==key) {
                value=t[id].value; return true;
            }
        }
        return false;
    }
};

int main() {
    int N; double L; cin>>N>>L;
    DoubleHashTable ht(N,L);
    string op,key; int value;
    while (cin>>op>>key) {
        if (op=="I") { cin>>value; ht.insert(key,value); }
        else if (op=="D") ht.erase(key);
        else { if (ht.search(key,value)) cout<<value<<'\n'; else cout<<"Not found\n"; }
    }

Exam 7 — Chaining rehash by maximum chain length (A2, 35 min)

Rules and implementation plan

  • Every 100 insertions পরে maxChain > C হলে প্রায় double size।
  • Every 100 deletions পরে maxChain < 0.8C হলে প্রায় half size, initial-এর নিচে নয়।
  • Rehash-এর আগে ও পরে average probe, load factor, max chain print। 10% live keys randomly search করে average probe।

Complete solution

#include <bits/stdc++.h>
using namespace std;

bool prime(int n){ if(n<2)return false; for(int i=2;1LL*i*i<=n;i++)if(n%i==0)return false; return true; }
int nextP(int n){ while(!prime(n))n++; return n; }
int prevP(int n){ while(n>=2&&!prime(n))n--; return n; }

struct Node { string key; int value; };

class ChainHash {
    vector<list<Node>> t;
    int initialSize, items=0, insSinceCheck=0, delSinceCheck=0;
    int C;
    mt19937 rng{42};

    int h(const string &s) const {
        unsigned long long x=0; for(unsigned char c:s)x=x*31+c;
        return x%t.size();
    }
    int maxChain() const {
        int ans=0; for(auto &b:t) ans=max(ans,(int)b.size()); return ans;
    }
    pair<bool,int> findWithProbe(const string &key) const {
        int probes=0;
        for(auto &x:t[h(key)]) { probes++; if(x.key==key)return {true,probes}; }
        return {false,probes};
    }
    vector<string> keys() const {
        vector<string> v; v.reserve(items);
        for(auto &b:t)for(auto &x:b)v.push_back(x.key);
        return v;
    }
    double averageProbe() {
        auto v=keys(); if(v.empty())return 0;
        shuffle(v.begin(),v.end(),rng);
        int take=max(1,(int)ceil(v.size()*0.10)), total=0;
        for(int i=0;i<take;i++)total+=findWithProbe(v[i]).second;
        return (double)total/take;
    }
    void report(const string &when) {
        cout<<fixed<<setprecision(3)<<when
            <<" | avgProbe="<<averageProbe()
            <<" load="<<(double)items/t.size()
            <<" maxChain="<<maxChain()<<'\n';
    }
    void insertRaw(const string &key,int value) {
        t[h(key)].push_back({key,value}); items++;
    }
    void rehash(int n) {
        report("Before"); auto old=move(t);
        t.assign(n,{}); items=0;
        for(auto &b:old)for(auto &x:b)insertRaw(x.key,x.value);
        report("After ");
    }
public:
    ChainHash(int N,int limit):initialSize(nextP(N)),C(limit){t.resize(initialSize);}
    void insert(const string &key,int value) {
        for(auto &x:t[h(key)])if(x.key==key){x.value=value;return;}
        insertRaw(key,value);
        if(++insSinceCheck==100) {
            insSinceCheck=0;
            if(maxChain()>C) rehash(nextP(2*t.size()));
        }
    }
    bool erase(const string &key) {
        auto &b=t[h(key)];
        for(auto it=b.begin();it!=b.end();++it)if(it->key==key){
            b.erase(it);items--;
            if(++delSinceCheck==100) {
                delSinceCheck=0;
                if(maxChain()<0.8*C && (int)t.size()>initialSize) {
                    int ns=max(initialSize,prevP(t.size()/2));
                    if(ns<t.size())rehash(ns);
                }
            }
            return true;
        }
        return false;
    }
};

    while(cin>>op>>key){if(op=="I"){cin>>value;ht.insert(key,value);}else ht.erase(key);}

New 6 — Backward-shift deletion without tombstones

Question

Linear probing table-এ tombstone ব্যবহার করা যাবে না। Delete-এর পরে following cluster-এর প্রয়োজনীয় keys backward shift করে search correctness বজায় রাখতে হবে।

Solution

int circularDistance(int home,int pos,int n){return (pos-home+n)%n;}

bool erase(const string &key){
    int n=table.size(),hole=-1;
    for(int i=0;i<n;i++){
        int id=(hash1(key)+i)%n;
        if(table[id].state==EMPTY)return false;
        if(table[id].key==key){hole=id;break;}
    }
    if(hole==-1)return false;

    int j=(hole+1)%n;
    while(table[j].state==OCCUPIED){
        int home=hash1(table[j].key)%n;
        if(circularDistance(home,j,n)>circularDistance(home,hole,n)){

    }
    table[hole]=Slot{};                    // final hole becomes EMPTY
    itemCount--;
    return true;
}

New 18 — Cuckoo hashing with offline Hash1 and Hash2

Question

দুটি table, প্রতিটির size prime N। Key হয় T1[h1] নয়তো T2[h2]-এ থাকবে। Collision হলে resident key kick করে alternate table-এ পাঠাও। 2N kicks-এর মধ্যে জায়গা না হলে size দ্বিগুণ prime করে rehash। Search O(1) worst-case দুই slot।

Solution core

struct Slot{string key;int value;bool used=false;};
vector<Slot>a,b;
vector<pair<string,int>> records;           // authoritative live data

bool build(int n){
    a.assign(n,{});b.assign(n,{});
    for(auto source:records){
        string key=source.first;int value=source.second,side=0;
        bool placed=false;
        for(int kick=0;kick<2*n;kick++){
            int id=(side==0?h1(key)%n:h2(key)%n);
            Slot &s=(side==0?a[id]:b[id]);
            if(!s.used){s={key,value,true};placed=true;break;}
            swap(key,s.key);swap(value,s.value);side^=1;
        }
        if(!placed)return false;             // caller retries from records
    }
    return true;
}

void insert(string key,int value){
    for(auto &p:records)if(p.first==key){
        p.second=value;                      // update authoritative record
        int n=a.size();while(!build(n))n=nextPrime(2*n);
        return;
    }
    records.push_back({key,value});
    int n=a.size();
    while(!build(n))n=nextPrime(2*n);        // cycle হলে clean rebuild
}

bool search(const string &key,int &value){
    auto &x=a[h1(key)%a.size()];if(x.used&&x.key==key){value=x.value;return true;}
    auto &y=b[h2(key)%b.size()];if(y.used&&y.key==key){value=y.value;return true;}
    return false;
}

Exam note: records authoritative রাখায় failed partial placement থেকেও কোনো key হারায় না। Delete support চাইলে record remove করে build(currentSize) করবে।
